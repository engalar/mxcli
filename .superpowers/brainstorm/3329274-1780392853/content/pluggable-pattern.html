发现 3：统一抽象 — PluggableWidget

所有插件 widget 可用一个泛型结构表示

type PluggableWidget struct {
    Name       string
    WidgetID   string              // "com.mendix.widget.custom.datagrid2.DataGrid2"
    DataSource *DataSourceDef         // nil for Image
    Slots      map[string][]Widget  // "filterWidgets", "content", "controlBar"
    Props      map[string]any         // 所有 Object.Properties 的 key-value
    RawObject  bson.D                 // 完整原始 Object（write-through 用）
}

关键：RawObject 保留 BSON 原始数据，写回时直接 write-through，不丢失 Properties 树

    w := &PluggableWidget{
        WidgetID:  extractWidgetID(doc),

func (c PluggableCodec) ToBSON(w Widget) bson.D {
    pw := w.(*PluggableWidget)
    doc := bson.D{...name, type...}
    // Write-through：原始 Object 完整保留
    doc = append(doc, bson.E{"Object", pw.RawObject})
    // 仅更新 slot widgets（如 filterWidgets）
    updateSlotInObject(pw.RawObject, pw.Slots)
    return doc
}

发现 5：插件族在 Codec Registry 中的位置
// mdl/canonical/page/widget/pluggable_codec.go — 一个 codec 覆盖所有插件 widget

type PluggableCodec struct{}

func (c PluggableCodec) Kinds() []types.WidgetKind {
    return []types.WidgetKind{
        types.WidgetDataGrid, types.WidgetGallery,
        types.WidgetComboBox, types.WidgetImage, types.WidgetUnknown,
    }
}

func (c PluggableCodec) LiftFromAST(w *ast.WidgetV3, _ string) (Widget, error) {
    // MDL 创建插件 widget 时，通过 AST 的 type 属性获取 widgetID
    return &PluggableWidget{
        Name:     w.Name,
        WidgetID: propStr(w, "widgetid"),
        // Props 来自 w.Properties（用户在 MDL 中显式设置的属性）
    }, nil
}

func (c PluggableCodec) HydrateFromBSON(doc bson.D) (Widget, error) {
    return &PluggableWidget{
        WidgetID:   extractCustomWidgetIDStr(doc),  // 复用现有函数
        DataSource: extractPluggableDataSource(doc), // 复用现有函数
        Slots:      extractNamedSlots(doc),          // 新：提取 filterWidgets/content 等
        RawObject:  dGetDoc(doc, "Object"),          // 新：保留完整 Object
    }, nil
}

func (c PluggableCodec) ToBSON(w Widget) bson.D {
    pw := w.(*PluggableWidget)
    updateSlotInObject(pw.RawObject, pw.Slots)  // 仅更新 slot 中的 widgets
    return bson.D{
        {"$Type", "CustomWidgets$CustomWidget"},
        {"Name", pw.Name},
        {"Object", pw.RawObject},  // write-through：属性树完整保留
    }
}

func (c PluggableCodec) ToMDL(w Widget, depth int, out io.Writer) {
    pw := w.(*PluggableWidget)
    // DataGrid/Gallery 的 MDL describe 由此函数生成
    kind := guessKindFromWidgetID(pw.WidgetID)   // datagrid / gallery / combobox / image
    renderPluggable(out, kind, pw, depth)
}

✓ 插件族的核心模式总结

1. 统一 BSON 格式（CustomWidgets$CustomWidget + Object.Properties）→ 一个 PluggableCodec 覆盖全部
2. WidgetID 区分具体 widget 类型 → ToMDL 按 ID pattern 渲染不同 MDL 语法
3. RawObject write-through → 完全消除当前 lossy 问题（Properties 树不再丢失）
4. Slots（filterWidgets/content）可单独更新，其余 Properties 原封保留
5. 未知插件 widget（WidgetUnknown）= PluggableWidget with WidgetID but no special ToMDL
