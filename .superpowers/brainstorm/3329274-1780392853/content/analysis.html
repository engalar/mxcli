推导出的正确设计：Widget Codec Registry

三个 switch 的根本问题：每个 widget 的知识分散在 3 个文件。正确方案：每个 widget 族有一个 Codec，包含该族所有 3 个操作。消费者通过 Registry 分发，不再写 switch。 

// mdl/canonical/page/widget_codec.go

type WidgetCodec interface {
    LiftFromAST(w *ast.WidgetV3, moduleName string) (Widget, error)
    HydrateFromGen(el element.Element) (Widget, []Warning, error)
    HydrateFromBSON(doc bson.D) (Widget, error)
    ToBSON(w Widget) bson.D
    ToMDL(w Widget, depth int, out io.Writer)
}

// 消费者不再写 switch：
func RenderWidget(registry *WidgetRegistry, w Widget, depth int, out io.Writer) {
    codec := registry.Lookup(w.Kind())
    codec.ToMDL(w, depth, out)   // ← 零 switch
}

func widgetToBSON(registry *WidgetRegistry, w Widget) bson.D {
    codec := registry.Lookup(w.Kind())
    return codec.ToBSON(w)         // ← 零 switch
}
// 族群 codec 文件结构：每族一文件，包含该族全部 3 个操作
