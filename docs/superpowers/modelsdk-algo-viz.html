// property/base.go — 所有 property 的公共底座
type propertyBase struct {
    name  string
    dirty bool
    owner *element.Base
    bit   uint  // 在父元素位图中的 bit 索引，全局唯一
}

func (b *propertyBase) Bind(owner *element.Base, bit uint) {
    b.owner = owner
    b.bit = bit
}

func (b *propertyBase) markDirty() {
    b.dirty = true
    if b.owner != nil {
        b.owner.MarkDirty(b.bit)  // 向上传播到元素位图
    }
}

// property/primitive.go — 泛型懒解码标量
type Primitive[T any] struct {
    propertyBase
    decode DecodeFunc[T]  // BSON 解码函数，由 codegen 选择
    raw    bson.Raw       // 借用引用，不拥有
    val    T
    loaded bool
}

func (p *Primitive[T]) Init(raw bson.Raw) { p.raw = raw }  // 绑定不解码

func (p *Primitive[T]) Get() T {
    if !p.loaded {
        if p.raw != nil {
            p.val = p.decode(p.raw, p.name)  // 首次访问才解码
        }
        p.loaded = true
    }
    return p.val
}

func (p *Primitive[T]) Set(v T) {
    p.val = v
    p.loaded = true
    p.markDirty()  // → owner.MarkDirty(bit) → 向上冒泡
}
// gen/domainmodels/types.go (代码生成产物)

// initAttribute — 解码器专用，属性已绑定但不标 dirty
func initAttribute() *Attribute {
    o := &Attribute{}
    o.SetTypeName("DomainModels$Attribute")

    o.name = property.NewPrimitive[string]("Name", property.DecodeString)
    o.name.Bind(&o.Base, 0)  // bit 0

    o.dataStorageGuid = property.NewPrimitive[string]("DataStorageGuid", property.DecodeString)
    o.dataStorageGuid.Bind(&o.Base, 1)  // bit 1

    o.propType = property.NewPart[element.Element]("Type")
    o.propType.Bind(&o.Base, 2)  // bit 2 (子元素)

    o.exportLevel = property.NewEnum[string]("ExportLevel")
    o.exportLevel.Bind(&o.Base, 6)

    o.SetProperties([]element.Property{
        o.name, o.dataStorageGuid, o.propType, o.exportLevel,
    })
    return o
}

// NewAttribute — 用户创建新元素，标 bit 63 = "全新元素"
func NewAttribute() *Attribute {
    o := initAttribute()
    o.MarkDirty(63)  // 强制 Encoder 做完整序列化（无 raw 可透传）
    return o
}

// ──────────────────────────────────────────────────
// 关键区别:
//   init*()  → 解码路径，dirty=false，保留 raw passthrough
//   New*()   → 创建路径，dirty=true，触发完整 buildDoc
// InitFromRaw — 在 Decoder.Decode() 调用 factory() 后立即执行
func (o *Attribute) InitFromRaw(raw bson.Raw) {
    // 标量：只绑定 raw，不做任何解码（懒加载）
    o.name.Init(raw)
    o.dataStorageGuid.Init(raw)
    o.documentation.Init(raw)

    // 子元素 (Part)：递归解码，子元素内部同样懒加载
    if child, err := codec.DecodeChild(raw, "Type"); err == nil {
        o.propType.SetFromDecode(child)  // SetFromDecode 不触发 markDirty
    }

    // 枚举：直接读取（字符串，无需懒加载）
    if val, err := raw.LookupErr("ExportLevel"); err == nil {
        if s, ok := val.StringValueOK(); ok {
            o.exportLevel.SetFromDecode(s)  // 同样不 dirty
        }
    }
}

// SetFromDecode vs Set 的区别:
//   Set(v)          → 标记 dirty，用于用户修改
//   SetFromDecode(v) → 不标 dirty，用于初始化（保留 raw passthrough）

// DecodeChild 内部实现
func DecodeChild(raw bson.Raw, key string) (element.Element, error) {
    val, err := raw.LookupErr(key)
    if err != nil { return nil, err }
    childRaw, ok := val.DocumentOK()
    if !ok { return nil, fmt.Errorf("not a document") }
    dec := NewDecoder(DefaultRegistry)
    return dec.Decode(childRaw)  // 递归，同样调 InitFromRaw
}
// element/element.go — 位图传播核心
type Base struct {
    dirty      []uint64  // 每个 uint64 容纳 64 个 bit
    childDirty bool     // 任意子孙被修改时为 true
    container  Element
}

func (b *Base) MarkDirty(bit uint) {
    word := bit / 64
    for len(b.dirty) <= int(word) {
        b.dirty = append(b.dirty, 0)
    }
    b.dirty[word] |= 1 << (bit % 64)

    // 向上传播：通知父元素有子孙变化
    if b.container != nil {
        if mc, ok := b.container.(interface{ MarkChildDirty() }); ok {
            mc.MarkChildDirty()
        }
    }
}

func (b *Base) MarkChildDirty() {
    if b.childDirty { return }  // 截断重复传播
    b.childDirty = true
    if b.container != nil {
        if mc, ok := b.container.(interface{ MarkChildDirty() }); ok {
            mc.MarkChildDirty()
        }
    }
}

func (b *Base) IsDirty() bool {
    if b.childDirty { return true }
    for _, w := range b.dirty {
        if w != 0 { return true }
    }
    return false
}

// 传播路径示意:
// attr.SetName("x")
//   → Primitive.Set → markDirty → attr.MarkDirty(0)
//     → entity.MarkChildDirty() → unit.MarkChildDirty()
func (e *Encoder) Encode(elem element.Element) ([]byte, error) {
    // 路径 A: 完全干净 → 零拷贝透传
    if elem.Raw() != nil && !elem.IsDirty() {
        return []byte(elem.Raw()), nil  // O(1)
    }
    doc, err := e.buildDoc(elem)
    if err != nil { return nil, err }
    return bson.Marshal(doc)
}

func (e *Encoder) buildDoc(elem element.Element) (bson.D, error) {
    var doc bson.D
    if raw := elem.Raw(); raw != nil {
        bson.Unmarshal(raw, &doc)  // 起点：保留所有未知字段
    } else {
        doc = bson.D{{"$ID", newID()}, {"$Type", elem.TypeName()}}
    }

    for _, prop := range elem.Properties() {
        wp := prop.(element.WritableProperty)

        // Part (单子元素) — 三路分支
        if cp, ok := prop.(element.ChildProperty); ok {
            child := cp.ChildElement()
            if wp.Dirty() || (child != nil && child.IsDirty()) {
                doc = setField(doc, prop.Name(), must(e.buildDoc(child)))
            }
            continue  // B3: 干净 → 保留 raw 字段
        }

        // PartList (子列表) — 三路分支
        if clp, ok := prop.(element.ChildListProperty); ok {
            if wp.Dirty() {
                arr := bson.A{int32(3)}  // C1: 结构变化 → 全量重建
                for _, c := range clp.ChildElements() {
                    arr = append(arr, must(e.buildDoc(c)))
                }
                doc = setField(doc, prop.Name(), arr)
            } else if anyChildDirty(clp) {
                arr := bson.A{int32(3)}  // C2: 选择性重建
                for _, c := range clp.ChildElements() {
                    if c.IsDirty() {
                        arr = append(arr, must(e.buildDoc(c)))
                    } else {
                        arr = append(arr, bson.Raw(c.Raw()))  // 干净子项透传
                    }
                }
                doc = setField(doc, prop.Name(), arr)
            }
            continue  // C3: 干净 → 保留 raw 字段
        }

        // 标量 — 只有 dirty 才覆写
        if !wp.Dirty() { continue }
        doc = setField(doc, prop.Name(), wp.BSONValue())
    }
    return doc, nil
}
// gen/domainmodels/types.go — package init() 注册所有类型
func init() {
    // 标准注册：qualified name == storage name
    codec.DefaultRegistry.Register("DomainModels$Attribute", func() element.Element {
        return initAttribute()
    })

    // 双注册处理历史别名：EntityImpl 是 BSON 存储名，Entity 是 SDK 公开名
    codec.DefaultRegistry.Register("DomainModels$EntityImpl", func() element.Element {
        return initEntity()
    })

    // 存储别名：DeleteBehavior → AssociationDeleteBehavior
    codec.DefaultRegistry.Register("DomainModels$DeleteBehavior", func() element.Element {
        return initAssociationDeleteBehavior()
    })
}

// TypeRegistry — 并发安全的工厂映射
type TypeRegistry struct {
    mu        sync.RWMutex
    factories map[string]func() element.Element
}

func (r *TypeRegistry) Lookup(typeName string) (func() element.Element, bool) {
    r.mu.RLock()
    f, ok := r.factories[typeName]
    r.mu.RUnlock()
    return f, ok
}

// 测试：标准 Roundtrip 验证三个不变式
func TestRoundtrip(t *testing.T) {
    dec := codec.NewDecoder(codec.DefaultRegistry)
    elem, _ := dec.Decode(original)

    // 不变式 1: 解码后必须 clean
    assert(!elem.IsDirty())

    enc := &codec.Encoder{}
    out, _ := enc.Encode(elem)

    // 不变式 2: clean → byte-identical
    assert(bytes.Equal(out, []byte(original)))

    attr := elem.(*domainmodels.Attribute)
    attr.SetName("Renamed")
    out2, _ := enc.Encode(attr)

    // 不变式 3: 未知字段必须保留
    assert(doc["Unknown"] == "preserved")
}
