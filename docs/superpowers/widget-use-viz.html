// mdl/executor/widget_engine.go

// WidgetDefinition — 从 *.def.json 加载，描述如何从 MDL 构建 widget
type WidgetDefinition struct {
    WidgetID         string             // "com.mendix.widget.web.combobox.Combobox"
    MDLName          string             // "combobox" — MDL 语句中使用的名称
    WidgetKind       string             // "pluggable" (React) | "custom" (Dojo)
    TemplateFile     string             // "combobox.json"
    DefaultEditable  string             // "Always" | "Never" | "Conditional"
    PropertyMappings []PropertyMapping  // 无模式时使用的全局映射
    ChildSlots       []ChildSlotMapping  // 子组件容器插槽
    Modes            []WidgetMode        // 条件模式列表（先匹配先用）
}

type PropertyMapping struct {
    PropertyKey string  // widget 属性 key（对应 PropertyType.Name）
    Source      string  // "Attribute" | "Association" | "DataSource" | ""
    Value       string  // 字面量值（primitive operation 时使用）
    Operation  string  // 10 种操作之一
    Default    string  // source 不存在时的回退值
}

type WidgetMode struct {
    Name             string             // "association" | "default"
    Condition        string             // "hasDataSource" | "hasAttribute" | ""
    PropertyMappings []PropertyMapping
    ChildSlots       []ChildSlotMapping
}

type ChildSlotMapping struct {
    PropertyKey  string  // widget 属性 key（Widgets 类型）
    MDLContainer string  // MDL 容器名："template" | "FILTER" 等
    Operation   string  // "widgets"
}
// Build() — PluggableWidgetEngine 的核心方法
func (e *PluggableWidgetEngine) Build(
    def *WidgetDefinition, w *ast.WidgetV3,
) (*backend.GenCustomWidgetElem, error) {

    // 保存/恢复 entityContext（DataSource 映射会改变它）
    oldCtx := e.pageBuilder.entityContext
    defer func() { e.pageBuilder.entityContext = oldCtx }()

    // Step 1: 加载模板 → builder 已含 BSON Type+Object+PropertyTypeIDs
    builder, err := e.backend.LoadWidgetTemplate(def.WidgetID, e.pageBuilder.getProjectPath())
    propertyTypeIDs := builder.PropertyTypeIDs()

    // Step 2: 选择模式（条件评估）→ PropertyMappings + ChildSlotMappings
    mappings, slots, err := e.selectMappings(def, w)

    // Step 3: 依次应用每个 PropertyMapping
    for _, mapping := range mappings {
        ctx, err := e.resolveMapping(mapping, w)       // MDL AST → BuildContext
        e.applyOperation(builder, mapping.Operation, mapping.PropertyKey, ctx)
    }

    // Step 3.5: DataSource 属性未配置时打印 HINT 日志（AI 友好提示）
    for _, m := range mappings {
        if m.Source == "DataSource" && w.GetDataSource() == nil {
            linked := linkedPropertyKeysFor(m.PropertyKey, propertyTypeIDs)
            log.Print("HINT: property '" + m.PropertyKey + "' not configured. Add DataSource: Module.Entity. Linked: " + linked)
        }
    }

    // Step 4: 自动 DataSource —— 未被 mapping 显式处理时，自动匹配第一个 DataSource 类型属性
    if !dsHandledByMapping {
        if ds := w.GetDataSource(); ds != nil {
            for propKey, entry := range propertyTypeIDs {
                if entry.ValueType == "DataSource" {
                    dataSource, entityName, _ := e.pageBuilder.buildDataSourceV3(ds)
                    builder.SetDataSourceOpaque(propKey, e.backend.SerializeGenElemToOpaque(dataSource))
                    e.pageBuilder.entityContext = entityName
                    break  // 只绑定第一个
                }
            }
        }
    }

    // Step 4.1: 应用子组件插槽
    e.applyChildSlots(builder, slots, w, propertyTypeIDs)

    // Step 5: Finalize — 填充必要的 ObjectList 默认值，返回最终 gen 元素
    return builder.Finalize()
}
// selectMappings — 条件模式选择
func (e *PluggableWidgetEngine) selectMappings(
    def *WidgetDefinition, w *ast.WidgetV3,
) ([]PropertyMapping, []ChildSlotMapping, error) {

    // 无模式 → 直接使用顶层映射
    if len(def.Modes) == 0 {
        return def.PropertyMappings, def.ChildSlots, nil
    }

    var fallback *WidgetMode
    var fallbackCount int

    for i := range def.Modes {
        mode := &def.Modes[i]
        if mode.Condition == "" {
            fallbackCount++
            if fallback == nil { fallback = mode }
            continue
        }
        // 条件求值：第一个匹配者立即返回
        if e.evaluateCondition(mode.Condition, w) {
            return mode.PropertyMappings, mode.ChildSlots, nil
        }
    }

    if fallback != nil {
        if fallbackCount > 1 {
            return nil, nil, fmt.Errorf("%d modes without condition; only one default allowed", fallbackCount)
        }
        return fallback.PropertyMappings, fallback.ChildSlots, nil
    }
    return nil, nil, fmt.Errorf("no matching mode for widget %s", def.MDLName)
}

// evaluateCondition — 内建条件枚举
func (e *PluggableWidgetEngine) evaluateCondition(cond string, w *ast.WidgetV3) bool {
    switch {
    case cond == "hasDataSource":  return w.GetDataSource() != nil
    case cond == "hasAttribute":   return w.GetAttribute() != ""
    case strings.HasPrefix(cond, "hasProp:"):
        return w.GetStringProp(strings.TrimPrefix(cond, "hasProp:")) != ""
    default: return false
    }
}
// applyOperation — 将 BuildContext 写入 WidgetObjectBuilder
func (e *PluggableWidgetEngine) applyOperation(
    builder backend.WidgetObjectBuilder,
    opName string,
    propKey string,
    ctx *BuildContext,
) error {
    switch opName {
    case "attribute":
        builder.SetAttribute(propKey, ctx.AttributePath)
    case "association":
        builder.SetAssociation(propKey, ctx.AssocPath, ctx.EntityName)
    case "primitive":
        builder.SetPrimitive(propKey, ctx.PrimitiveVal)
    case "selection":
        builder.SetSelection(propKey, ctx.PrimitiveVal)
    case "expression":
        builder.SetExpression(propKey, ctx.PrimitiveVal)
    case "datasource":
        builder.SetDataSourceOpaque(propKey, ctx.DataSource)
    case "widgets":
        // 子组件不走此路径 — 由 applyChildSlots 处理
    case "texttemplate":
        builder.SetTextTemplate(propKey, ctx.PrimitiveVal)
    case "action":
        builder.SetActionOpaque(propKey, ctx.Action)
    case "attributeObjects":
        builder.SetAttributeObjects(propKey, ctx.AttributePaths)
    default:
        return fmt.Errorf("unknown operation %q for property %s", opName, propKey)
    }
    return nil
}
// WidgetRegistry — 三级加载与查找
type WidgetRegistry struct {
    byMDLName       map[string]*WidgetDefinition  // uppercase MDLName → def
    byWidgetID      map[string]*WidgetDefinition  // widgetId → def
    knownOperations map[string]bool
    projectDir      string
    mpkNameMap      map[string]string            // MDLName → widgetID (MPK 兜底)
}

// NewWidgetRegistryWithOps — 初始化时按三级顺序加载
func NewWidgetRegistryWithOps(extraOps map[string]bool) (*WidgetRegistry, error) {
    r := &WidgetRegistry{
        byMDLName:  make(map[string]*WidgetDefinition),
        byWidgetID: make(map[string]*WidgetDefinition),
        knownOperations: mergeOps(defaultKnownOperations, extraOps),
    }
    // Tier 3: 内建嵌入（最先加载，优先级最低）
    if err := r.loadFromFS(definitions.FS); err != nil { return nil, err }

    // Tier 2 + 1: 用户全局 → 项目级（后加载覆盖先加载）
    if home, err := os.UserHomeDir(); err == nil {
        r.loadFromDir(filepath.Join(home, ".mxcli/widgets"))
    }
    r.loadFromDir(filepath.Join(r.projectDir, ".mxcli/widgets"))

    return r, nil
}

// Lookup — MDL 名称查找
func (r *WidgetRegistry) Lookup(mdlName string) *WidgetDefinition {
    return r.byMDLName[strings.ToUpper(mdlName)]
}
