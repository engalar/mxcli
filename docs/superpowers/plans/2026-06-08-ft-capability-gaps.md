```go
func (v *Visitor) EnterSynchronizeStatement(ctx *parser.SynchronizeStatementContext) {
	s := &ast.SynchronizeStmt{}
	if ctx.VARIABLE() != nil {
		// Strip leading '$' from the variable token
		raw := ctx.VARIABLE().GetText()
		if len(raw) > 0 && raw[0] == '$' {
			raw = raw[1:]
		}
		s.Variable = raw
	}
	v.pushStatement(s)
}
```

```go
// SPDX-License-Identifier: Apache-2.0

package executor

import (
	"github.com/mendixlabs/mxcli/mdl/ast"
	"github.com/mendixlabs/mxcli/modelsdk/element"
	genMf "github.com/mendixlabs/mxcli/modelsdk/gen/microflows"
)

// addSynchronizeActionGen adds a Microflows$SynchronizeAction to the nanoflow.
//
//	synchronize $Order;  → Type="SelectedObjects", VariableNames="Order"
//	synchronize;         → Type="All", VariableNames=""
func (fb *flowBuilderGen) addSynchronizeActionGen(s *ast.SynchronizeStmt) element.ID {
	action := genMf.NewSynchronizeAction()
	assignFreshID(action)
	action.SetErrorHandlingType(fb.ehTypeGen(nil))

	if s.Variable != "" {
		action.SetType("SelectedObjects")
		action.SetVariableNames(s.Variable)
	} else {
		action.SetType("All")
		action.SetVariableNames("")
	}

	return fb.genActivityWrap(action, nil, "")
}
```

```go
// createNavigationProfileViaModelsdk inserts a new navigation profile document.
// profileType: "NativeMobile" → Navigation$NativeMobileNavigationProfile
// profileName: the display name in Studio Pro (e.g. "OfflineNative")
// containerID: the project's navigation container (from GetNavigation().ID)
func (b *MprBackend) createNavigationProfileViaModelsdk(profileType, profileName string, containerID model.ID) error {
    if b.msdkWriter == nil {
        return fmt.Errorf("modelsdk writer not initialized")
    }

    bsonType := "Navigation$WebNavigationProfile"
    switch profileType {
    case "NativeMobile":
        bsonType = "Navigation$NativeMobileNavigationProfile"
    }

    newID := modelsdkmpr.GenerateID()
    contents, err := bson.Marshal(bson.D{
        {Key: "$ID", Value: modelsdkmpr.IDToBsonBinary(newID)},
        {Key: "$Type", Value: bsonType},
        {Key: "Name", Value: profileName},
        {Key: "HomePage", Value: nil},
        {Key: "LoginPage", Value: nil},
        {Key: "NotFoundPage", Value: nil},
        {Key: "RoleBasedHomePages", Value: bson.A{int32(3)}},
        {Key: "MenuItemCollection", Value: bson.D{
            {Key: "$ID", Value: modelsdkmpr.IDToBsonBinary(modelsdkmpr.GenerateID())},
            {Key: "$Type", Value: "Navigation$MenuItemCollection"},
            {Key: "Items", Value: bson.A{int32(3)}},
        }},
        {Key: "ExcludedRoles", Value: bson.A{int32(1)}},
    })
    if err != nil {
        return fmt.Errorf("marshal navigation profile: %w", err)
    }

    return b.insertUnit(newID, string(containerID), "Profiles", bsonType, contents)
}
```
